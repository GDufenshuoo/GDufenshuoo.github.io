模拟架构管理及高通量计算

background ​

在之前的模拟中，总是存在诸如温度点扫描之类的许多类似的独立模拟过程，其中文件管理基本处于一种随性所欲的状态，产生了诸多诸如删错样本，重复提交任务，忘了模拟进度，以及读不懂到底做了什么又放在哪里这样的那样的十分不便的事。于是乎，在老师项目管理方法的启发下，计划借助大火的DeepSeek，基于DrWatson.jl快速构建一套模拟架构，顺便实现一下高通量计算。

julia
# src/config.jl
using DrWatson: @strdict, savename

struct SimulationParameters
    system::Symbol
    software::Symbol
    temperature::Float64
    timestep::Float64
    steps::Int
    forcefield::String
    topology_file::String
end

function DrWatson.savename(p::SimulationParameters)
    base = savename((;
        software = p.software,
        system = p.system,
        T = round(p.temperature, digits=1),
        dt = p.timestep
    ), connector = "-", equals = "")
    return base * "_$(p.forcefield)"
end

function load_config(filename)
    wload(simdir("configs", filename))
end

# scripts/generate_htc.jl
using DrWatson, Distributed
@quickactivate :MolecularSimulationHT

include(srcdir("paths.jl"))
include(srcdir("config.jl"))

const FF_VERSIONS = Dict(
    "martini3" => 3.0,
    "charmm36" => 36.2
)

function generate_all_jobs()
    base_params = Dict(
        :system => [:protein_folding, :membrane_formation],
        :software => [:hoomd, :lammps],
        :temperature => collect(280:20:360),
        :timestep => [0.001, 0.002],
        :steps => 1_000_000,
        :forcefield => ["martini3", "charmm36"],
        :topology_file => ["protein.top", "membrane.top"]
    )

    configs = dict_list(base_params)
    
    # 并行生成任务文件
    @distributed for config in configs
        sp = SimulationParameters(; config...)
        generate_input_files(sp)
        generate_slurm_script(sp)
    end
    
    tmpsave(configs, simdir("configs"))
end

function generate_input_files(sp::SimulationParameters)
    if sp.software == :hoomd
        template = read(resdir("hoomd_scripts", "template.py"), String)
        filled = replace(template,
            "{{TEMPERATURE}}" => sp.temperature,
            "{{TIMESTEP}}" => sp.timestep,
            "{{STEPS}}" => sp.steps,
            "{{FORCEFIELD}}" => sp.forcefield
        )
        write(simdir("data", "inputs", savename(sp)*".py"), filled)
    end

end

# src/simulator.jl
using DrWatson: produce_or_load, @tag!

function run_simulation(config_name::String)
    config = load_config(config_name)
    
    result_path = produce_or_load(simdir("data", "results"), config) do
        result = execute_simulation(config)
        @tag!(result)
        result["performance"] = @timed execute_simulation(config)
        return result
    end
    
    postprocessing(result_path)
end

function execute_simulation(config)
    if config.software == :hoomd
        input_file = simdir("data", "inputs", savename(config)*".py")
        run(`hoomd -n 8 $input_file`)
        return parse_hoomd_output(config)
    end
    # 其他软件执行逻辑...
end

6. 分析系统增强版

julia
# simulations/analysis/analyze.jl
using DrWatson, DataFrames, Statistics

function collect_simulation_results()
    df = collect_results(
        simdir("data", "results"),
        black_list = [:snapshots, :performance],
        special_list = [
            :final_energy => d -> d.results.energy[end],
            :mean_rdf => d -> mean(d.results.rdf),
            :walltime => d -> d.performance.time
        ]
    )
    
    transform!(df, 
        :walltime => ByRow(t -> t/3600) => :walltime_hours,
        :temperature => ByRow(round ∘ Int) => :temp_rounded
    )
    
    safesave(simdir("analysis", "summary.jld2"), df)
    return df
end

# Papers/generate_figures.jl
using DrWatson, Plots, LaTeXStrings

function generate_paper_figures()
    df = wload(simdir("analysis", "summary.jld2"))[:df]
    
    # Figure 1: 温度效应
    plt1 = @df df plot(
        :temperature, :final_energy,
        group=:software, 
        xlabel=L"Temperature (K)", 
        ylabel=L"Energy (k$_B$T)",
        legend=:topright
    )
    
    # Figure 2: 计算效率
    plt2 = @df df boxplot(
        :software, :walltime_hours,
        xlabel="Software", 
        ylabel="Wall Time (hours)"
    )
    
    savefig(plt1, plotdir("figure1.pdf"))
    savefig(plt2, plotdir("figure2.pdf"))
end

8. 监控系统实现

julia
# simulations/monitor.jl
using DrWatson, TerminalLoggers

function job_monitor(; refresh_interval=60)
    @info "Starting job monitoring..."
    while true
        try
            df = collect_simulation_results()
            completed = nrow(df)
            total = length(readdir(simdir("configs")))
            
            println("""
            Job Progress: ($completed/$total)
            Recent failures: $(count(ismissing.(df.walltime)))
            Estimated completion: $(round((total - completed)*mean(df.walltime_hours), digits=1)) hours
            """)
            
            sleep(refresh_interval)
        catch e
            @error "Monitoring error" exception=(e, catch_backtrace())
            break
        end
    end
end

# src/simulator.jl 新增
function adaptive_scheduler()
    # 根据历史性能数据动态调整任务分配
    perf_data = collect_performance_data()
    scheduler = PriorityScheduler()
    
    for job in pending_jobs()
        priority = calculate_priority(job, perf_data)
        schedule!(scheduler, job, priority)
    end
    return scheduler
end

# scripts/generate_htc.jl 新增
function validate_forcefield(ff_name)
    ff_path = resdir("force_fields", ff_name)
    required_files = ["forcefield.itp", "molecule.top"]
    
    if !all(isfile.(joinpath.(ff_path, required_files)))
        @error "Forcefield $ff_name incomplete!"
        return false
    end
    return true
end
